# hello_mics: 4-mic PDM array -> pio1 -> free-running DMA ring -> CIC.
# A 1 kHz tone reaches only mic C (rightmost), so only its meter moves.
set mics 0 0 1 0          # pickup: mic A, B, C, D
set tone 1000             # level defaults to 8000 (a comment must not zero it)
expect "mic C: rms= 1[34][0-9][0-9]" 20000   # the meters once the filters settle
screenshot out/hello_mics.png
quit
